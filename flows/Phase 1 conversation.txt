from executor.tool_registry import ACTION_REGISTRY
import time

class Executor:
    def __init__(self, delay: float = 1.0):
        self.registry = ACTION_REGISTRY
        self.delay = delay

    def validate_action(self, action: dict):
        if "action" not in action:
            raise ValueError("Missing 'action' field")

        if action["action"] not in self.registry:
            raise ValueError(f"Unknown action: {action['action']}")

    def execute_step(self, action: dict):
        self.validate_action(action)

        action_name = action["action"]
        func = self.registry[action_name]

        # Remove 'action' key before passing args
        params = {k: v for k, v in action.items() if k != "action"}

        try:
            func(**params)
            return {"status": "success", "action": action_name}
        except Exception as e:
            return {
                "status": "error",
                "action": action_name,
                "message": str(e)
            }

    def run(self, plan: list):
        results = []

        for step in plan:
            result = self.execute_step(step)
            results.append(result)

            print(f"[EXECUTOR] {result}")

            if result["status"] == "error":
                print("[EXECUTOR] Stopping due to failure.")
                break

            time.sleep(self.delay)

        return results

import json
import re

def clean_llm_output(output: str) -> str:
    # Remove markdown code blocks like ```json ... ```
    output = re.sub(r"```json\s*", "", output)
    output = re.sub(r"```", "", output)

    # Trim whitespace
    return output.strip()


def parse_llm_output(output: str):
    try:
        cleaned = clean_llm_output(output)
        return json.loads(cleaned)
    except Exception as e:
        raise ValueError(f"Invalid JSON from LLM:\n{output}")

class ConversationState:
    def __init__(self):
        self.pending_intent = None
        self.missing_field = None

    def set_pending(self, intent: dict, missing_field: str):
        self.pending_intent = intent
        self.missing_field = missing_field

    def clear(self):
        self.pending_intent = None
        self.missing_field = None

    def is_waiting(self):
        return self.pending_intent is not None

from intent.slot_schema import INTENT_SCHEMA

def find_missing_slot(intent: dict):
    intent_name = intent.get("intent")

    if intent_name not in INTENT_SCHEMA:
        return None

    required_fields = INTENT_SCHEMA[intent_name]

    for field in required_fields:
        if not intent.get(field):
            return field

    return None


def generate_question(missing_field: str):
    questions = {
        "app": "Which application do you want to open?",
        "text": "What should I type?",
        "key": "Which key should I press?"
    }

    return questions.get(missing_field, f"Please provide {missing_field}")

from llm.llm_client import call_gemini
import json
import re

SYSTEM_PROMPT = """
Extract intent and parameters from user input.

Return JSON only.

Available intents:
- open_app (app)
- type_text (text)
- press_key (key)

Example:
User: open chrome
Output:
{"intent": "open_app", "app": "chrome"}
"""

def clean_output(output):
    output = re.sub(r"```json\s*", "", output)
    output = re.sub(r"```", "", output)
    return output.strip()

def extract_intent(user_input: str):
    prompt = SYSTEM_PROMPT + f"\nUser: {user_input}\nOutput:"
    
    raw = call_gemini(prompt)
    cleaned = clean_output(raw)

    return json.loads(cleaned)

from executor.executor import Executor
from intent.intent_extractor import extract_intent
from clarification.clarifier import find_missing_slot, generate_question
from clarification.conversation_state import ConversationState
from planner.planner import create_plan

def main():
    executor = Executor(delay=1)
    state = ConversationState()

    while True:
        user_input = input("\nEnter command: ")

        if user_input.lower() in ["exit", "quit"]:
            break

        try:
            # 🔁 If waiting for clarification
            if state.is_waiting():
                intent = state.pending_intent
                intent[state.missing_field] = user_input
                state.clear()
            else:
                intent = extract_intent(user_input)

            print("\n[INTENT]")
            print(intent)

            # 🔍 Check missing slot
            missing = find_missing_slot(intent)

            if missing:
                question = generate_question(missing)
                print(f"\n[CLARIFICATION] {question}")
                state.set_pending(intent, missing)
                continue

            # 📦 Create plan
            plan = create_plan(intent)

            print("\n[PLAN]")
            print(plan)

            # ⚙️ Execute
            executor.run(plan)

        except Exception as e:
            print(f"[ERROR] {str(e)}")

if __name__ == "__main__":
    main()

class ConversationState:
    def __init__(self):
        self.pending_plan = None
        self.issue = None

    def set_pending(self, plan, issue):
        self.pending_plan = plan
        self.issue = issue

    def clear(self):
        self.pending_plan = None
        self.issue = None

    def is_waiting(self):
        return self.pending_plan is not None

from executor.executor import Executor
from planner.plan_generator import generate_plan
from planner.validator import validate_plan

from clarification.conversation_state import ConversationState

def main():
    executor = Executor(delay=1)
    state = ConversationState()

    while True:
        user_input = input("\nEnter command: ")

        if user_input.lower() in ["exit", "quit"]:
            break

        try:
            # 🔁 If waiting for clarification
            if state.is_waiting():
                # Inject missing info into last step
                for step in state.pending_plan:
                    if state.issue == "missing_text":
                        step["text"] = user_input
                    elif state.issue == "missing_app":
                        step["app"] = user_input
                    elif state.issue == "missing_key":
                        step["key"] = user_input

                plan = state.pending_plan
                state.clear()

            else:
                plan = generate_plan(user_input)

            print("\n[PLAN]")
            print(plan)

            # 🔍 Validate plan
            issue = validate_plan(plan)

            if issue:
                question = generate_question(issue)
                print(f"\n[CLARIFICATION] {question}")

                state.set_pending(plan, issue)
                continue

            # ⚙️ Execute
            executor.run(plan)

        except Exception as e:
            print(f"[ERROR] {str(e)}")

from executor.executor import Executor

from planner.plan_generator import generate_plan
from planner.validator import validate_plan

from critic.reviewer import review_plan

from clarification.conversation_state import ConversationState
from clarification.resume_handler import merge_clarification


def main():

    executor = Executor(delay=1)
    state = ConversationState()

    while True:

        user_input = input("\nEnter command: ").strip()

        if user_input.lower() in ["exit", "quit"]:
            break

        try:

            # -------------------------------
            # Handle clarification continuation
            # -------------------------------
            if state.awaiting_clarification:

                reconstructed_request = merge_clarification(
                    state.pending_original_request,
                    state.pending_question,
                    user_input
                )

                print(
                    "\n[RECONSTRUCTED REQUEST]"
                )
                print(reconstructed_request)

                state.clear()

                # use reconstructed request
                user_input = reconstructed_request


            # -------------------------------
            # Planner
            # -------------------------------
            plan = generate_plan(user_input)

            print("\n[PLAN]")
            print(plan)


            # -------------------------------
            # Deterministic Validation
            # -------------------------------
            issue = validate_plan(plan)

            if issue:
                print(
                    f"\n[VALIDATION ERROR] {issue}"
                )
                continue


            # -------------------------------
            # Critic Review
            # -------------------------------
            critic_result = review_plan(
                user_input,
                plan
            )

            print("\n[CRITIC]")
            print(critic_result)


            # -------------------------------
            # Clarification Trigger
            # -------------------------------
            if critic_result.startswith("CLARIFY"):

                question = (
                    critic_result
                    .replace("CLARIFY:", "")
                    .strip()
                )

                print(
                    f"\n[CLARIFICATION] {question}"
                )

                state.set_pending(
                    user_input,
                    question
                )

                continue


            # -------------------------------
            # Execute
            # -------------------------------
            if critic_result.strip() == "EXECUTE":

                print("\n[EXECUTING PLAN]")
                executor.run(plan)

            else:
                print(
                  "\n[ERROR] Unexpected critic response"
                )


        except Exception as e:
            print(f"\n[ERROR] {str(e)}")


if __name__ == "__main__":
    main()

class ConversationState:

    def __init__(self):
        self.awaiting_clarification=False
        self.pending_original_request=None
        self.pending_question=None
        self.clarification_count=0


    def set_pending(
        self,
        request,
        question
    ):
        self.awaiting_clarification=True
        self.pending_original_request=request
        self.pending_question=question


    def clear(self):
        self.awaiting_clarification=False
        self.pending_original_request=None
        self.pending_question=None
        self.clarification_count=0

from executor.executor import Executor

from classifier.ambiguity_checker import (

from planner.plan_generator import generate_plan

from critic.reviewer import review_plan

from clarification.conversation_state import (
    ConversationState
)

from clarification.resume_handler import (
    merge_clarification
)


MAX_CLARIFICATIONS=2


def main():

    executor=Executor(delay=1)

    state=ConversationState()


    while True:

        user_input=input(
            "\nEnter command: "
        ).strip()


        if user_input.lower() in [
            "exit",
            "quit"
        ]:
            break


        try:

            # ------------------------
            # clarification continuation
            # ------------------------
            if state.awaiting_clarification:

                user_input=merge_clarification(
                    state.pending_original_request,
                    state.pending_question,
                    user_input
                )

                print(
                  "\n[RECONSTRUCTED REQUEST]"
                )
                print(user_input)


            # ------------------------
            # ambiguity classifier
            # ------------------------
            ambiguity_result=check_ambiguity(
                user_input
            )

            print(
                "\n[INTERPRETER]"
            )
            print(
                ambiguity_result
            )


            if ambiguity_result.startswith(
                "AMBIGUOUS"
            ):

                state.clarification_count +=1

                if (
                   state.clarification_count
                   >=MAX_CLARIFICATIONS
                ):

                    print(
                     "\nToo much ambiguity."
                    )
                    print(
                     "Please rephrase request."
                    )

                    state.clear()

                    continue


                question=(
                    ambiguity_result
                    .replace(
                      "AMBIGUOUS:",
                      ""
                    )
                    .strip()
                )


                print(
                    "\n[CLARIFICATION]"
                )
                print(question)


                state.set_pending(
                    user_input,
                    question
                )

                continue


            # resolved -> clear pending
            if state.awaiting_clarification:
                state.clear()



            # ------------------------
            # planner
            # ------------------------
            plan=generate_plan(
                user_input
            )

            print("\n[PLAN]")
            print(plan)



            # ------------------------
            # deterministic validation
            # ------------------------
            issue=validate_plan(
                plan
            )

            if issue:
                print(
                  f"[VALIDATION ERROR] {issue}"
                )
                continue



            # ------------------------
            # critic review
            # ------------------------
            critic_result=review_plan(
                user_input,
                plan
            )

            print(
                "\n[CRITIC]"
            )
            print(
               critic_result
            )


            if (
              critic_result.strip()
              =="EXECUTE"
            ):

                executor.run(
                    plan
                )

            else:
                print(
                 "Critic rejected plan."
                )


        except Exception as e:

            print(
              f"\n[ERROR] {e}"
            )


if __name__=="__main__":
    main()

import psutil
import pygetwindow as gw


TARGET_APPS={
 "chrome.exe":"chrome",
 "Code.exe":"vscode",
 "notepad.exe":"notepad"
}


def get_running_apps():

    apps=[]

    for proc in psutil.process_iter(
       ['name']
    ):
        try:
            name=proc.info['name']

            if name in TARGET_APPS:
                apps.append(
                  TARGET_APPS[name]
                )

        except:
            continue

    return list(
       set(apps)
    )


def get_system_state():

    try:
        win=gw.getActiveWindow()

        title=(
            win.title
            if win
            else None
        )

        return {
          "active_window":title,
          "running_apps":
             get_running_apps()
        }

    except:
        return {
          "active_window":None,
          "running_apps":[]
        }

import win32gui
import win32process
import psutil


APP_PROCESS = {
   "chrome":"chrome.exe",
   "vscode":"Code.exe",
   "notepad":"notepad.exe"
}


def focus_app(app):

    target=APP_PROCESS.get(
       app.lower()
    )

    if not target:
        raise Exception(
          "unknown app"
        )


    target_pids=[]


    for proc in psutil.process_iter(
      ['pid','name']
    ):
        try:
            if (
              proc.info["name"]
              == target
            ):
                target_pids.append(
                    proc.info["pid"]
                )
        except:
            pass



    if not target_pids:
        raise Exception(
           f"{app} not running"
        )


    def enum_handler(hwnd,_):

        if not win32gui.IsWindowVisible(
            hwnd
        ):
            return

        _,pid=win32process.GetWindowThreadProcessId(
            hwnd
        )

        if pid in target_pids:

            win32gui.ShowWindow(
                hwnd,
                5
            )

            win32gui.SetForegroundWindow(
                hwnd
            )

            raise StopIteration


    try:
        win32gui.EnumWindows(
            enum_handler,
            None
        )

    except StopIteration:
        return {
          "status":"success",
          "action":"focus_app"
        }


    raise Exception(
       "No window found"
    )
